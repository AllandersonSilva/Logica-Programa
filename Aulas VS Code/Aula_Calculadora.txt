import java.util.Scanner;

public class Calculator {

    static double somar(double numero01, double numero02) {
        return numero01 + numero02;
    }
    static double subtrair(double numero01, double numero02) {
        return numero01 - numero02;
    }
    static double multiplicar(double numero01, double numero02) {
        return numero01 * numero02;
    }
    static double dividir(double numero01, double numero02) {
        return numero01 / numero02;   
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("digite o primeiro numero: ");
        double numero1 = sc.nextDouble();

        System.out.println("Digite o segundo numero: ");
        double numero2 = sc.nextDouble();

        System.out.println("Soma: " + somar(numero1, numero2));
        System.out.println("Subtracao: " + subtrair(numero1, numero2));
        System.out.println("Multiplicacao: " + multiplicar(numero1, numero2));
        
        if (numero2 != 0) {
            System.out.println("Divisão: " + dividir(numero1, numero2));
        } else {
            System.out.println("Divisão: Não  possivel dividir por zero!");
        }
    
        sc.close();
            }
        }
